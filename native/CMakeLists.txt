cmake_minimum_required(VERSION 3.20)
project(MediaHubBridge LANGUAGES CXX)
add_executable(mediahub-bridge bridge.cpp)
set_target_properties(mediahub-bridge PROPERTIES OUTPUT_NAME "mediahub-pipe")
target_compile_features(mediahub-bridge PRIVATE cxx_std_17)
target_compile_definitions(mediahub-bridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mediahub-bridge PRIVATE advapi32)
if(MSVC)
  set_property(TARGET mediahub-bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(mediahub-bridge PRIVATE /W4 /WX /EHsc)
endif()
install(TARGETS mediahub-bridge RUNTIME DESTINATION .)
